const config = {
    type: Phaser.AUTO,
    width: 800,
    height: 450,
    parent: 'game-container',
    physics: {
        default: 'arcade',
        arcade: { gravity: { y: 800 }, debug: false }
    },
    scene: { preload: preload, create: create, update: update }
};

const game = new Phaser.Game(config);
let player, cursors, interactKey, msgText, item;

function preload() {
    // IMPORTANTE: Asegúrate de que la imagen se llame exactamente protagonista.png
    this.load.image('protagonista', 'protagonista.png');
    
    // Usamos un fondo de color sólido o una imagen si la tienes
    this.load.image('suelo', 'https://labs.phaser.io/assets/sprites/platform.png');
    this.load.image('recuerdo', 'https://labs.phaser.io/assets/sprites/apple.png');
}

function create() {
    // Fondo gris oscuro estilo guerra/histórico
    this.cameras.main.setBackgroundColor('#333');

    // Suelo (plataforma invisible o visible)
    let platforms = this.physics.add.staticGroup();
    let ground = platforms.create(400, 440, 'suelo').setScale(3).refreshBody();
    ground.setTint(0x222222); // Oscurecer el suelo

    // Personaje (La mujer estilo Toyotaro)
    player = this.physics.add.sprite(100, 300, 'protagonista');
    player.setCollideWorldBounds(true);
    player.setScale(0.3); // Ajusta este valor si la imagen es muy grande

    // Objeto interactuable (El recuerdo)
    item = this.physics.add.staticImage(650, 400, 'recuerdo');

    // Texto de interfaz (Estilo narrativo)
    msgText = this.add.text(400, 100, '', { 
        fontSize: '22px', 
        fill: '#ffffff',
        fontFamily: 'Arial',
        backgroundColor: '#00000088',
        padding: { x: 10, y: 5 }
    }).setOrigin(0.5);

    // Controles
    cursors = this.input.keyboard.createCursorKeys();
    interactKey = this.input.keyboard.addKey(Phaser.Input.Keyboard.KeyCodes.E);

    // Colisiones
    this.physics.add.collider(player, platforms);
}

function update() {
    // Movimiento Lateral
    if (cursors.left.isDown) {
        player.setVelocityX(-180);
        player.flipX = true; // Espejo de imagen
    } else if (cursors.right.isDown) {
        player.setVelocityX(180);
        player.flipX = false;
    } else {
        player.setVelocityX(0);
    }

    // Lógica de cercanía (Distancia entre personaje y objeto)
    let dist = Phaser.Math.Distance.Between(player.x, player.y, item.x, item.y);

    if (item.active && dist < 80) {
        msgText.setText('Presiona [E] para investigar');
        if (Phaser.Input.Keyboard.JustDown(interactKey)) {
            msgText.setText('Un viejo diario... narra una historia de esperanza.');
            item.destroy(); // El objeto desaparece al recogerlo
            
            // El mensaje se borra después de 4 segundos
            this.time.delayedCall(4000, () => {
                msgText.setText('');
            });
        }
    } else {
        // Borra el mensaje si te alejas sin interactuar
        if (msgText.text === 'Presiona [E] para investigar') {
            msgText.setText('');
        }
    }
}
